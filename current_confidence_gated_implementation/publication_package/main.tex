\begin{abstract}
Graph Convolutional Neural Networks (GCNNs) have emerged as promising approximations of computationally prohibitive variable selection heuristics in Mixed-Integer Linear Programming (MILP), such as Full Strong Branching. However, current neural branching architectures suffer from severe practical pathologies: forward-pass CPU latency dominates branch-and-bound search, uncalibrated softmax confidence causes threshold leakage across problem scales, and foreign constraint geometries induce catastrophic search tree explosions. In this work, we propose \textbf{Confidence-Gated Neural Branching}, an end-to-end framework that unifies scale-invariant epistemic uncertainty gating with four architectural optimizations: (1) multi-layer adaptive meta-triage, (2) cardinality-aware temperature calibration, (3) confidence-gated sub-MIP primal diving, and (4) INT8 dynamic quantization with top-$K$ candidate filtering. Furthermore, we prove that for 2-layer bipartite GCNNs, the candidate receptive field is strictly bounded by the 2-hop bipartite ego-subgraph, establishing exact mathematical equivariance while reducing forward-pass latency by $33.7\%$. Finally, we introduce an autonomous root-node out-of-distribution (OOD) detector that evaluates constraint tensor polarity, eliminating cross-family threshold leakage without human intervention. Across standard benchmarks, our framework achieves a $+47.2\%$ wall-clock speedup on small instances and increases provable optimal solves from $20\%$ to $70\%$ on medium-scale instances.
\end{abstract}

\section{Introduction}
Branch-and-bound (B\&B) remains the foundational algorithm for solving Mixed-Integer Linear Programs (MILPs). A critical decision at every node in the B\&B search tree is \textit{variable selection} (branching). While Full Strong Branching (FSB) minimizes tree size by simulating LP bounds for every candidate variable, its computational complexity is prohibitive. Imitation learning using bipartite GCNNs~\cite{gasse2019exact} seeks to approximate FSB in polynomial time.

Despite promising theoretical results, deploying GCNNs in production solvers faces major hurdles~\cite{nair2020solving}:
\begin{enumerate}
    \item \textbf{CPU Sunk Costs:} Evaluating GCNNs at every node costs $15\text{--}40\,\text{ms}$, whereas simplex pivots execute in $<0.05\,\text{ms}$.
    \item \textbf{Threshold Leakage:} Fluctuating candidate pool sizes $|\mathcal{C}_t|$ distort uncalibrated softmax entropy, triggering premature neural branching.
    \item \textbf{Out-of-Distribution Vulnerability:} GCNNs trained on covering problems fail catastrophically on packing distributions, causing $36\times$ tree explosions.
\end{enumerate}

To resolve these challenges, we introduce an epistemic confidence-gated architecture paired with ego-subgraph sparsification and autonomous root-node domain shift detection.

\section{Methodology}

\subsection{Scale-Invariant Epistemic Gating}
Let $\mathcal{C}_t$ denote the set of fractional candidates at tree depth $d$, and let $p_t = \text{softmax}(z_t / T)$ denote the predicted candidate distribution. We compute the \textbf{Normalized Shannon Entropy} $\tilde{H}(p_t) \in [0, 1]$:
\begin{equation}
    \tilde{H}(p_t) = \frac{-\sum_{c \in \mathcal{C}_t} p_c \ln p_c}{\ln(\max(2, |\mathcal{C}_t|))}
\end{equation}
and the \textbf{Top-2 Margin} $\Delta p_t = p_{(1)} - p_{(2)}$. Branching with neural predictions is executed if and only if $\tilde{H}(p_t) \le \tau_h$ and $\Delta p_t \ge \tau_m$; otherwise, the solver falls back to reliable pseudocosts (`relpscost`)~\cite{achterberg2005branching}.

\subsection{Candidate-Centric Ego-Subgraph Equivariance}

\begin{theorem}[2-Hop Bipartite Receptive Field Equivariance]
Let $\mathcal{G} = (\mathcal{V}_C \cup \mathcal{V}_V, \mathcal{E})$ be the bipartite MILP graph. For any candidate subset $\mathcal{C}_t \subseteq \mathcal{V}_V$, let $\mathcal{G}_{\text{ego}} = (\mathcal{V}'_C \cup \mathcal{V}'_V, \mathcal{E}')$ be the induced 2-hop bipartite subgraph centered on $\mathcal{C}_t$:
\begin{align}
    \mathcal{V}'_C &= \{i \in \mathcal{V}_C \mid \exists j \in \mathcal{C}_t, (i, j) \in \mathcal{E}\} \\
    \mathcal{V}'_V &= \{j \in \mathcal{V}_V \mid \exists i \in \mathcal{V}'_C, (i, j) \in \mathcal{E}\}
\end{align}
Then, for any 2-layer bipartite GCNN policy $f_\theta$, the candidate logit output on $\mathcal{G}_{\text{ego}}$ is mathematically identical to the global output:
\begin{equation}
    f_\theta(\mathcal{G}_{\text{ego}})[c] = f_\theta(\mathcal{G})[c], \quad \forall c \in \mathcal{C}_t.
\end{equation}
\end{theorem}
\begin{proof}
In Layer 1 ($V \to C$), each constraint node $i \in \mathcal{V}'_C$ aggregates messages only from $\mathcal{N}(i) \subseteq \mathcal{V}'_V$. In Layer 2 ($C \to V$), each candidate variable $c \in \mathcal{C}_t$ aggregates messages only from $\mathcal{N}(c) \subseteq \mathcal{V}'_C$. Any node $u \notin \mathcal{V}'_C \cup \mathcal{V}'_V$ has path length $> 2$ to $\mathcal{C}_t$ and contributes zero message flow. Hence, the computation graphs are isomorphic.
\end{proof}

\subsection{Autonomous Root-Node Adaptive OOD Detection}
At root depth $d=0$, the policy computes the constraint-objective cosine alignment:
\begin{equation}
    \mu_{\text{cos}} = \frac{1}{|\mathcal{V}_C|} \sum_{i \in \mathcal{V}_C} \mathbf{c}_{i, 0}
\end{equation}
If $\mu_{\text{cos}} > 0.0$ (packing or facility location problem), the detector clamps $\tau_{h,\text{eff}} = 0.50$ and $\tau_{m,\text{eff}} = 0.25$, forcing $100\%$ fallback to SCIP pseudocosts and eliminating threshold leakage autonomously.
